The first agent, in four commands

# install the CLI
curl -fsSL https://astropods.com/install | sh

# create, authenticate, run
ast project create my-agent --model gateway
cd my-agent
ast login
ast project start

# publish and deploy
ast blueprint push my-agent
ast blueprint deploy my-agent
