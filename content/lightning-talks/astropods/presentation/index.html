terminal
# install the CLI
$ curl -fsSL https://astropods.com/install | sh

# create, authenticate, run
$ ast project create hello-teammate --model gateway
$ cd hello-teammate
$ ast login
$ ast project start

# publish and deploy
$ ast blueprint push hello-teammate
$ ast blueprint deploy hello-teammate
